int calcular_pontuacao(float dados_paciente[]){ //definindo a pontuação de cada fator//
    int pontos = 0;
    
    if (dados_paciente[0] >= 65){   //idade//
        pontos += 3;
    }
    if (dados_paciente[1] >= 160){   //pressao//
        pontos += 3;
    }
    if (dados_paciente[2] == 1){     //tabagismo//
        pontos += 3;
    }
    if (dados_paciente[3] == 1){    //diabetes//
        pontos += 2;
    }
    if (dados_paciente[4] == 1){    //cardiaca//
        pontos += 2;
    }
    if (dados_paciente[5] >= 160){   //colesterol//
        pontos += 2;
    }
    
    //Calcular imc//
    float peso = dados_paciente[6]; 
    float altura = dados_paciente[7];

    float imc = peso / (altura * altura);
    
    if (imc >= 30){
        pontos += 2;
    }
    
    
    if (dados_paciente[8] == 1){    //sedentarismo//
        pontos += 1;
    }
    
    if (dados_paciente[9] == 1){    //alimentação//
        pontos += 1;
    }
    
    if (dados_paciente[10] == 1){    //estresse//
        pontos += 1;
    }
    if (dados_paciente[11] == 1){    //sono//
        pontos += 1;
    }
    if (dados_paciente[12] == 1){    //sexo masculino//
        pontos += 1;
    }
    
    return pontos;
}

void classificacao_risco(int pontos){ //Verifica o total de pontos e classifica o paciente//
    if (pontos >= 0 && pontos <= 4){
        printf("Classificação: Menor de 10%\n");
    } else{
        if (pontos >= 5 && pontos <= 9){
            printf("Classificação: Entre 10%% e 20%%\n");
        }else {
            if (pontos >= 10 && pontos <= 14){
                printf("Classificação: Entre 20%% e 30%\n");
            }else{
                if (pontos >= 15 && pontos <= 18){
                   printf("Classificação: Entre 30%% e 50%\n"); 
                }else{
                    printf("Classificação: Acima de 50%%\n");
                }
            }
        }
    }
}
