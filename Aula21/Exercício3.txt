#include <stdio.h>

typedef struct {
    char nome[50];
    float salario;
    int horasExtras;
} Funcionario;

int main(void){
    
    FILE *arquivo;
    
    Funcionario f;
    int quantidade;
    
    printf("===== CADASTRO DE FUNCIONÁRIOS ===== \n");
    printf("\n");
    
    printf("Quantos funcionários deseja cadastrar? ");
    scanf("%d", &quantidade);
    
    if (quantidade < 1 || quantidade > 5){
        printf("Quantidade inválida!\n");
        return 0;
    }
    
    arquivo = fopen("funcionarios.txt", "a");
    
    if (arquivo == NULL){
        printf("Erro ao abrir arquivo!\n");
        return 1;
    }
    
    for (int i = 0; i < quantidade; i++){
        
        printf("\nFuncionario %d\n", i + 1);
        
        printf("Nome: ");
        scanf(" %49[^\n]", f.nome);
        
        printf("Salário: ");
        scanf("%f", &f.salario);
        
        printf("Horas extras: ");
        scanf("%d", &f.horasExtras);
        
        fprintf(arquivo, "%s;%.2f;%d\n", f.nome, f.salario, f.horasExtras);
        
    }
    
    fclose(arquivo);
    
    printf("\nFuncionários cadastrados com sucesso!\n");
    
    arquivo = fopen("funcionarios.txt", "r");
    
    if (arquivo == NULL){
        printf("Erro ao ler arquivo!\n");
        return 1;
    }
    
    printf("\n");
    printf("===== FUNCIONÁRIOS CADASTRADOS ===== \n");
    
    while (fscanf(arquivo, " %49[^;];%f;%d", f.nome, &f.salario, &f.horasExtras) == 3){
        
        printf("\nNome: %s\n", f.nome);
        printf("Salário: R$ %.2f\n", f.salario);
        printf("Horas extras: %d\n", f.horasExtras);
        
    }
    
    fclose(arquivo);
    
    return 0;
    
}
