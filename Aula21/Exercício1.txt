#include <stdio.h>

int main(void){
    
    FILE *arquivo;
    
    char nome[50];
    int matricula;
    int quantidade;
    
    printf("===== CADASTRO DE FUNCIONÁRIOS ===== \n");
    printf("\n");
    
    printf("Quantos funcionários deseja cadastrar? ");
    scanf("%d", &quantidade);
    
    if (quantidade < 1 || quantidade > 5){
        printf("Quantidade inválida!\n");
        return 0;
    }
    
    arquivo = fopen("funcionarios.txt", "w");
    
    if (arquivo == NULL){
        printf("Erro ao abrir arquivo!\n");
        return 1;
    }
    
    for (int i = 0; i < quantidade; i++){
        
        printf("\nFuncionário %d\n", i + 1);
        
        printf("Nome: ");
        scanf(" %49[^\n]", nome);
        
        printf("Matrícula: ");
        scanf("%d", &matricula);
        
        fprintf(arquivo, "Nome: %s | Matrícula: %d\n", nome, matricula);
        
    }
    
    fclose(arquivo);
    
    printf("\nDados gravados com sucesso!\n");
    printf("Arquivo: funcionarios.txt\n");
    
    return 0;
    
}
